// Museum Card Game (MCG) - Chargement des cartes et images

let allCards = [];
let cardImages = {};

// Charger les cartes
fetch('src/data/cards.json')
    .then(response => response.json())
    .then(data => {
        allCards = [
            ...data.monuments || [],
            ...data.oeuvres || [],
            ...data.artistes || [],
            ...data.evenements || []
        ];
        console.log('Cartes chargées:', allCards.length);
        // Ajouter les URLs d'images
        loadCardImages();
    })
    .catch(error => {
        console.error('Erreur de chargement des cartes:', error);
        allCards = getDefaultCards();
        loadCardImages();
    });

// Charger les images des cartes
function loadCardImages() {
    fetch('src/data/cardImages.json')
        .then(response => response.json())
        .then(data => {
            cardImages = data.images || {};
            console.log('Images des cartes chargées:', Object.keys(cardImages).length);
            // Mettre à jour les cartes avec les URLs d'images
            allCards.forEach(card => {
                if (!card.image || card.image.endsWith('.png') || card.image.includes('assets/')) {
                    card.image = cardImages[card.id] || data.fallback;
                }
            });
        })
        .catch(error => {
            console.error('Erreur de chargement des images:', error);
            cardImages = {};
        });
}
